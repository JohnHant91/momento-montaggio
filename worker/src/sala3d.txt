function canvasTex(w, h, fn) {
  const c = document.createElement('canvas'); c.width = w; c.height = h; fn(c.getContext('2d'), w, h);
  const t = new THREE.CanvasTexture(c); t.colorSpace = THREE.SRGBColorSpace; t.anisotropy = 4; return t;
}
const texLegno = () => canvasTex(512, 128, (g, w, h) => {

  g.fillStyle = gr; g.fillRect(0, 0, w, h);
  for (let i = 0; i < 60; i++) {
    g.strokeStyle = 'rgba(150,105,55,' + (0.03 + Math.random() * 0.07) + ')'; g.lineWidth = 1 + Math.random() * 2;
    g.beginPath(); const y = Math.random() * h; g.moveTo(0, y);
    for (let x = 0; x <= w; x += 32) g.lineTo(x, y + Math.sin(x / 70 + i) * 3);
    g.stroke();
  }
});
function righeTesto(g, testo, x, y, larg, alt, maxR) {
  const parole = String(testo || '').split(/\s+/); let riga = '', n = 0;
  for (const p of parole) {
    const prova = riga ? riga + ' ' + p : p;
    if (g.measureText(prova).width > larg && riga) { g.fillText(riga, x, y + n * alt); riga = p; if (++n >= maxR) return n; }
    else riga = prova;
  }
  if (riga) { g.fillText(riga, x, y + n * alt); n++; }
  return n;
}

// ogni personaggio Mixamo ha il suo prefisso sulle ossa (mixamorig, mixamorig1, mixamorig9…): rinomino le tracce delle animazioni
function clipPer(M, prefisso) {
  if (M.perPrefisso[prefisso]) return M.perPrefisso[prefisso];
  const out = {};
  for (const [n, c] of Object.entries(M.clip)) {
    const k = c.clone();
    for (const tr of k.tracks) tr.name = tr.name.replace(/^mixamorig\d*/, prefisso);
    out[n] = k;
  }
  return (M.perPrefisso[prefisso] = out);
}

function creaGLB(id, M) {
  const h = hash(id), donna = DONNE.has(id);
  const lista = (donna ? DONNE_M : UOMINI).filter((n) => M.corpi[n]);
  const base = M.corpi[lista[h % lista.length]] || Object.values(M.corpi)[0];
  const modello = SkeletonUtils.clone(base.scene);
  modello.scale.setScalar(base.scala);
  modello.traverse((m) => { if (m.isMesh) { m.castShadow = true; m.frustumCulled = false; } });
  const o = new THREE.Group(), w = new THREE.Group(); w.add(modello); o.add(w);
  w.rotation.y = Math.PI; w.position.z = SEDUTA_DZ; // i personaggi Mixamo guardano +z, la scrivania è verso -z
  const mixer = new THREE.AnimationMixer(modello);
  const C = clipPer(M, base.prefisso);
  const az = {
    scrive: mixer.clipAction(C.Typing), fermo: mixer.clipAction(C.SittingIdle || C.Typing),
    Walk: mixer.clipAction((donna && C.FemaleWalk) || C.StandardWalk), Idle: mixer.clipAction(C.StandingIdle),
  };
  for (const a of Object.values(az)) a.time = Math.random() * 3;
  return { o, w, modello, mixer, az, corrente: null, seduto: true, dz: SEDUTA_DZ };
}
function glbVai(G, nome, dissolvi = 0.4) {
  if (G.corrente === nome) return;
  const nuova = G.az[nome]; nuova.enabled = true; nuova.setEffectiveWeight(1); nuova.play();
  if (G.corrente) G.az[G.corrente].crossFadeTo(nuova, dissolvi, false); else nuova.reset().play();
  G.corrente = nome;
}

  const anim = ['Typing', 'SittingIdle', 'StandardWalk', 'FemaleWalk', 'StandingIdle'].filter((n) => pronti.includes('anim_' + n));
  if (!anim.includes('Typing') || !anim.includes('StandardWalk') || !anim.includes('StandingIdle')) return;
  const piccolo = innerWidth < 700; // da telefono: un uomo e una donna soli, per non appesantire
  const corpi = [...UOMINI.slice(0, piccolo ? 1 : 4), ...DONNE_M.slice(0, piccolo ? 1 : 4)].filter((n) => pronti.includes(n));
  if (!corpi.length) return;
  const L = new GLTFLoader();
  const M = { corpi: {}, clip: {}, perPrefisso: {} };
  await Promise.all([
    ...anim.map(async (n) => { const g = await L.loadAsync('/persone/anim_' + n + '.glb'); M.clip[n] = g.animations[0]; }),
    ...corpi.map(async (n) => {
      const g = await L.loadAsync('/persone/' + n + '.glb');
      g.scene.traverse((o) => { if (o.isBone && /Hips$/.test(o.name) && !g.prefisso) g.prefisso = o.name.slice(0, -4); });
      g.prefisso ||= 'mixamorig';
      // altezza dalle ossa (testa - piede): il riquadro delle mesh a volte è sballato
      g.scene.updateMatrixWorld(true);
      const osso = (n) => g.scene.getObjectByName(g.prefisso + n);
      const testa = osso('HeadTop_End') || osso('Head'), piede = osso('LeftToeBase') || osso('LeftFoot');
      let alto = testa && piede ? testa.getWorldPosition(new THREE.Vector3()).y - piede.getWorldPosition(new THREE.Vector3()).y + 0.06 * (testa.getWorldPosition(new THREE.Vector3()).y) : 0;
      if (!(alto > 0)) { const b = new THREE.Box3().setFromObject(g.scene); alto = b.max.y - b.min.y; }
      g.scala = ALTEZZA / Math.max(1e-3, alto);
      M.corpi[n] = g;
    }),
  ]);
  if (!S) return;
  for (const id in S.agenti) {
    const P = S.agenti[id];
    const G = creaGLB(id, M);
    G.o.position.copy(P.U.o.position); G.o.rotation.y = P.U.o.rotation.y;
    G.o.traverse((m) => { m.userData.ag = id; });
    S.scena.remove(P.U.o); S.scena.add(G.o);
    P.G = G; P.U = { o: G.o };
    if (P.cammino) glbMuovi(G, 'Walk', 0); else glbSeduto(G, 0, P.stato === 'lavora', 0);
  }
  S.personeVere = true;
}

function posaSeduto(U, t, lavora) {
  U.fianchi.position.y = 0.52; U.busto.rotation.x = 0.06;
  for (const [i, g] of U.gambe.entries()) { g.anca.rotation.set(Math.PI / 2 - 0.08, 0, (i ? 1 : -1) * 0.05); g.ginocchio.rotation.x = -(Math.PI / 2 - 0.12); }
  for (const [i, b] of U.braccia.entries()) {
    const k = lavora ? Math.sin(t * 13 + i * 1.7) * 0.06 : 0;
    b.spalla.rotation.set(0.5 + k, 0, (i ? -1 : 1) * 0.1); b.gomito.rotation.set(1.0 - k, 0, 0);
  }
  U.testa.rotation.set(lavora ? 0.06 : -0.02, lavora ? Math.sin(t * 0.7) * 0.1 : Math.sin(t * 0.3) * 0.25, 0);
}
function posaCammino(U, fase) {
  const s = Math.sin(fase), c = Math.cos(fase);
  U.fianchi.position.y = 0.92 + Math.abs(c) * 0.025; U.busto.rotation.x = 0.04;
  U.gambe[0].anca.rotation.set(s * 0.42, 0, 0); U.gambe[1].anca.rotation.set(-s * 0.42, 0, 0);
  U.gambe[0].ginocchio.rotation.x = -(Math.max(0, -s) * 0.8 + 0.05); U.gambe[1].ginocchio.rotation.x = -(Math.max(0, s) * 0.8 + 0.05);
  U.braccia[0].spalla.rotation.set(-s * 0.38, 0, 0.06); U.braccia[1].spalla.rotation.set(s * 0.38, 0, -0.06);
  U.braccia[0].gomito.rotation.x = 0.2 + Math.max(0, -s) * 0.3; U.braccia[1].gomito.rotation.x = 0.2 + Math.max(0, s) * 0.3;
  U.testa.rotation.set(0, 0, 0);
}

function giroDiSedie(cx, cz, rx, rz, n, stanza) {
  const posti = [];
  for (let i = 0; i < n; i++) {
    const a = (i / n) * Math.PI * 2 + Math.PI / n, se = sedia(S.mat);
    const x = cx + Math.cos(a) * rx, z = cz + Math.sin(a) * rz;
    se.position.set(x, 0, z); se.rotation.y = Math.atan2(cx - x, cz - z) + Math.PI; S.scena.add(se);
    const d = new THREE.Vector3(cx - x, 0, cz - z).normalize();
    posti.push({ pos: new THREE.Vector3(x + d.x * 0.12, 0, z + d.z * 0.12), rot: Math.atan2(-d.x, -d.z) });
  }
  (S.posti ||= {})[stanza] = posti;
}

function costruisci(dati) {

  const pav = box(LARGO, 0.04, FONDO, mat.pav, 0.05); pav.position.set(CX, 0.01, CZ); pav.receiveShadow = true; scena.add(pav);
  // corridoio: parquet chiaro con un tappeto lungo
  const cor = box(LARGO, 0.03, COR_D, mat.legno, 0.01); cor.position.set(CX, 0.04, COR_Z + COR_D / 2); cor.receiveShadow = true; scena.add(cor);
  const tappeto = new THREE.Mesh(new RoundedBoxGeometry(LARGO - 1.2, 0.012, 0.8, 2, 0.006), new THREE.MeshStandardMaterial({ color: 0xd8d2c6, roughness: 0.95 }));
  tappeto.position.set(CX, 0.062, COR_Z + COR_D / 2); tappeto.receiveShadow = true; scena.add(tappeto);
  for (const x of [1.0, LARGO / 2, LARGO - 1.0]) { const a = agave(mat, 0.75); a.position.set(x, 0.05, COR_Z + 0.35); scena.add(a); }
  // pareti: alte sul fondo, basse davanti (vista "casa delle bambole"), porta sul corridoio
  const muro = (x0, z0, x1, z1, h, porta) => {
    const lungo = Math.hypot(x1 - x0, z1 - z0), orizz = Math.abs(z1 - z0) < 0.01;
    const pezzi = porta ? [[0, porta - 0.55], [porta + 0.55, lungo]] : [[0, lungo]];
    for (const [s0, s1] of pezzi) {
      if (s1 - s0 < 0.05) continue;
      const m = box(orizz ? s1 - s0 : 0.12, h, orizz ? 0.12 : s1 - s0, mat.bianco, 0.02);
      const c = (s0 + s1) / 2; m.position.set(orizz ? x0 + c : x0, h / 2, orizz ? z0 : z0 + c); scena.add(m);
    }
    if (porta) { // stipiti in legno
      for (const d of [-0.58, 0.58]) { const st = box(orizz ? 0.07 : 0.14, h + 0.04, orizz ? 0.14 : 0.07, mat.bordo, 0.01); st.position.set(orizz ? x0 + porta + d : x0, (h + 0.04) / 2, orizz ? z0 : z0 + porta + d); scena.add(st); }
    }
  };
  const ALTO = 1.55, BASSO = 0.38;
  muro(0, 0, LARGO, 0, 2.3); // parete di fondo dell'edificio
  S.aree = {}; S.agenti = {}; S.etAree = [];
  const uffici = Object.fromEntries(dati.uffici.map((u) => [u.id, u]));
  for (const R of STANZE) {
    // pavimento della stanza
    const fl = new THREE.Mesh(new RoundedBoxGeometry(R.w - 0.14, 0.014, R.d - 0.14, 2, 0.006), new THREE.MeshStandardMaterial({ color: new THREE.Color(TINTE[R.id] || '#f3f3f1').offsetHSL(0, 0.15, -0.03), roughness: 0.6 }));
    fl.position.set(R.x + R.w / 2, 0.04, R.z + R.d / 2); fl.receiveShadow = true; fl.userData.uff = R.id; scena.add(fl);
    // pareti laterali (alte verso il fondo, basse verso chi guarda) e parete sul corridoio con la porta
    const xs = [R.x, R.x + R.w];
    for (const x of xs) {
      if (R.lato === 'n') { muro(x, 0, x, R.d * 0.55, ALTO); muro(x, R.d * 0.55, x, R.d, BASSO + 0.25); }
      else { muro(x, R.z, x, R.z + R.d * 0.45, ALTO - 0.4); muro(x, R.z + R.d * 0.45, x, R.z + R.d, BASSO); }
    }
    const porta = R.porta || Math.min(R.w - 0.8, R.lato === 'n' ? R.w - 1.0 : 1.0);
    if (R.lato === 'n') muro(R.x, R.z + R.d, R.x + R.w, R.z + R.d, BASSO, porta);

    if (R.id === 'meeting') { arredaMeeting(R); continue; }
    if (R.id === 'direzione') arredaDirezione(R);

    const x0 = R.x + (R.w - largo) / 2, z0 = R.scrivanie != null ? R.z + R.scrivanie : R.z + Math.max(0.9, (R.d - righe * FILA) / 2 + 0.25);

        const mon = monitor(mat, A, R.id === 'video' ? 'video' : ''); mon.g.position.set(x, 0.785, zDesk - 0.2); mon.g.rotation.x = -0.22; scena.add(mon.g);
        const tast = box(0.42, 0.015, 0.13, mat.nero, 0.005); tast.position.set(x, 0.79, zDesk + 0.12); scena.add(tast);
        const se = sedia(mat); se.position.set(x, 0, zDesk + 0.72); scena.add(se);
        const U = umano(A.id); U.o.position.set(x, 0, zDesk + 0.62); scena.add(U.o);
        [mon.g, se, U.o].forEach((o) => o.traverse((m) => { m.userData.ag = A.id; }));
        S.agenti[A.id] = { A, U, x, z: zDesk + 0.62, casa: new THREE.Vector3(x, 0, zDesk + 0.62), sm: mon.sm, tex: mon.tex, ctx: mon.ctx, tipo: R.id === 'video' ? 'video' : '', stanza: R, seduto: true };
      }
    }
    // una pianta per stanza
    const pi = agave(mat, 0.85); pi.position.set(R.x + R.w - 0.45, 0.05, R.lato === 'n' ? R.z + 0.5 : R.z + R.d - 0.5); scena.add(pi);

  const tv = box(3.0, 1.6, 0.08, mat.nero, 0.02); tv.position.set(ST('video').x + 4.6, 1.35, 0.1); scena.add(tv);

export function aggiorna(dati) {
  if (!S) return;
  S.dati = dati;
  const t = performance.now() / 1000;
  for (const A of dati.agenti) {
    const P = S.agenti[A.id]; if (!P) continue;
    if (P.msg !== A.msg) P.t0 = t;
    P.stato = A.stato; P.msg = A.msg; P.log = A.log || []; P.A = { ...P.A, ...A };
    if (A.img && P.imgSrc !== A.img) { P.imgSrc = A.img; P.img = new Image(); P.img.onload = () => disegnaSchermo(P, performance.now() / 1000); P.img.src = A.img; }
    if (!A.img) { P.img = null; P.imgSrc = null; }
    P.sm.emissiveIntensity = P.stato === 'lavora' || P.stato === 'coda' ? 0.85 : 0.25;
    disegnaSchermo(P, t);
  }
  aggiornaRiunioni(dati);
  disegnaEtichette();
}

const esc = (s) => String(s || '').replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
function disegnaEtichette() {
  const el = S.etichette; let h = '';
  for (const E of S.etAree) h += '<div class="et-uff" data-uff="' + E.id + '">' + esc(E.nome) + '</div>';
  for (const A of S.dati.agenti) {
    if (!S.agenti[A.id]) continue;
    const viva = A.stato === 'lavora' || A.stato === 'coda' || S.scelto === A.id || !!S.agenti[A.id]?.riunione;
    h += viva
      ? '<div class="et-ag carta ' + (A.stato || '') + (S.scelto === A.id ? ' scelto' : '') + '" data-ag="' + A.id + '"><b>' + esc(A.n) + '</b><span>' + esc((A.msg || A.r || '').slice(0, 46)) + '</span><i>' + (A.stato === 'coda' ? 'In coda' : 'Guarda ▸') + '</i></div>'
      : '<div class="et-ag pillola ' + (A.stato || '') + '" data-ag="' + A.id + '"><span class="pt"></span>' + esc(A.n) + '</div>';
  }
  el.innerHTML = h;
  el.querySelectorAll('[data-ag]').forEach((x) => x.onclick = (ev) => { ev.stopPropagation(); vaiAgente(x.dataset.ag); });
  el.querySelectorAll('[data-uff]').forEach((x) => x.onclick = (ev) => { ev.stopPropagation(); entraUfficio(x.dataset.uff); });
  S.nodiAg = [...el.querySelectorAll('.et-ag')]; S.nodiUff = [...el.querySelectorAll('.et-uff')];
}

const tmp = new THREE.Vector3();
function schermo(v, W, H) { tmp.copy(v).project(S.camera); return [(tmp.x + 1) / 2 * W, (1 - tmp.y) / 2 * H, tmp.z]; }
function posiziona() {
  const r = S.renderer.domElement.getBoundingClientRect(), W = r.width, H = r.height;
  const dist = S.camera.position.distanceTo(S.controls.target);
  S.etichette.classList.toggle('vicino', dist < S.distTutto * 0.62);
  const scala = Math.min(1.1, Math.max(0.68, Math.sqrt(S.distTutto / dist) * 0.72));
  for (const el of S.nodiAg || []) {
    const P = S.agenti[el.dataset.ag]; if (!P) continue;
    const [x, y] = schermo(tmp.set(P.U.o.position.x, P.seduto ? 1.5 : 1.95, P.U.o.position.z), W, H);
    el.style.transform = 'translate(' + x.toFixed(1) + 'px,' + y.toFixed(1) + 'px) translate(-50%,-100%) scale(' + scala.toFixed(3) + ')';
    el.style.zIndex = String(Math.round(y));
  }
  S.etAree.forEach((E, i) => {
    const el = S.nodiUff[i]; if (!el) return;
    const [x, y] = schermo(E.pos, W, H);
    el.style.transform = 'translate(' + x.toFixed(1) + 'px,' + y.toFixed(1) + 'px) translate(-50%,0)';
  });
}

// ---------------------------------------------------------------- camera (prospettiva, di fronte e dall'alto)
const DIREZIONE = new THREE.Vector3(0, 1.25, 1).normalize();
function distanzaPer(w, d) {
  const vfov = THREE.MathUtils.degToRad(S.camera.fov), hfov = 2 * Math.atan(Math.tan(vfov / 2) * S.camera.aspect);
  const dw = (w / 2) / Math.tan(hfov / 2), dd = (d * 0.62 + 1.2) / 2 / Math.tan(vfov / 2);
  return Math.max(dw, dd) * 1.08;
}
function vola(target, dist, ms = 1000) {
  const dir = S.camera.position.clone().sub(S.controls.target).normalize();
  S.volo = { t0: performance.now(), ms, da: S.controls.target.clone(), a: target.clone(), d0: S.camera.position.distanceTo(S.controls.target), d1: dist, dir0: dir, dir1: dir.clone().lerp(DIREZIONE, 0.6).normalize() };
}
export function grandangolo(subito) {
  S.scelto = null; disegnaEtichette();
  const t = new THREE.Vector3(CX, 0, CZ - 0.4), d = distanzaPer(LARGO + 1.0, FONDO + 1.2);
  S.distTutto = d;
  if (subito) { S.controls.target.copy(t); S.camera.position.copy(t).addScaledVector(DIREZIONE, d); S.camera.lookAt(t); }
  else { vola(t, d); S.volo.dir1 = DIREZIONE.clone(); }
  S.cb.vista && S.cb.vista('tutto');
}
export function entraUfficio(u) {
  const A = S.aree[u]; if (!A) return;
  S.scelto = null; disegnaEtichette();
  vola(new THREE.Vector3(A.x + A.w / 2, 0.4, A.z + A.d / 2 + 0.3), distanzaPer(A.w + 0.8, A.d + 0.8));
  S.cb.ufficio && S.cb.ufficio(u);
}
export function vaiAgente(a) {
  const P = S.agenti[a]; if (!P) return;
  S.scelto = a; disegnaEtichette();
  const t = new THREE.Vector3(P.U.o.position.x, 0.9, P.U.o.position.z - 0.35);
  if (innerWidth < 700) t.z += 0.9;
  vola(t, distanzaPer(3.4, 2.6));
  S.cb.agente && S.cb.agente(a);
}

// ---------------------------------------------------------------- gente che cammina
// chi è libero ogni tanto si alza: esce dalla porta, percorre il corridoio, va alla macchinetta o in un'altra stanza, torna
function porta(R) {
  const px = R.x + (R.porta || Math.min(R.w - 0.8, R.lato === 'n' ? R.w - 1.0 : 1.0));
  return R.lato === 'n' ? { dentro: new THREE.Vector3(px, 0, R.z + R.d - 0.7), fuori: new THREE.Vector3(px, 0, COR_Z + COR_D * 0.35) }
    : { dentro: new THREE.Vector3(px, 0, R.z + 0.7), fuori: new THREE.Vector3(px, 0, COR_Z + COR_D * 0.65) };
}
function percorso(P) {
  const da = porta(P.stanza), stanze = STANZE.filter((r) => r.id !== P.stanza.id);
  const meta = Math.random() < 0.55 ? STANZE.find((r) => r.id === 'relax') : stanze[Math.floor(Math.random() * stanze.length)];
  const a = porta(meta);
  const alzati = new THREE.Vector3(P.casa.x + 0.55, 0, P.casa.z + 0.35);
  const fondo = meta.id === 'relax' ? new THREE.Vector3(meta.x + 0.9, 0, meta.z + 1.1) : a.dentro.clone().add(new THREE.Vector3(0, 0, meta.lato === 'n' ? -0.6 : 0.6));
  const andata = [alzati, new THREE.Vector3(alzati.x, 0, da.dentro.z), da.dentro, da.fuori, a.fuori, a.dentro, fondo];
  return { punti: [...andata, 'sosta', ...andata.slice(0, -1).reverse(), P.casa.clone()], i: 0, sosta: 0 };
}
function stanzaDi(pos) { return STANZE.find((r) => pos.x > r.x && pos.x < r.x + r.w && pos.z > r.z && pos.z < r.z + r.d) || null; }
function viaggio(da, aStanza, fine) {
  const pts = [], dS = stanzaDi(da);
  if (dS && dS !== aStanza) { const d = porta(dS); pts.push(new THREE.Vector3(da.x, 0, d.dentro.z), d.dentro, d.fuori); }
  if (aStanza && dS !== aStanza) { const a = porta(aStanza); pts.push(a.fuori, a.dentro); }
  pts.push(fine);
  return pts;
}
function vaiInRiunione(P, Q, i) {
  const posti = S.posti?.[Q.stanza]; if (!posti) return;
  const seat = posti[i % posti.length], o = P.U.o;
  const parti = P.cammino ? o.position.clone() : new THREE.Vector3(P.casa.x + 0.55, 0, P.casa.z + 0.35);
  const R = STANZE.find((r) => r.id === Q.stanza);
  P.riunione = Q.id;
  P.cammino = { punti: [parti, ...viaggio(parti, R, seat.pos)], i: 0, sosta: 0, fine: { tipo: 'riunione', seat } };
  P.seduto = false; P.fase = 0;
}
function tornaDaRiunione(P) {
  P.riunione = null;
  const o = P.U.o, alzati = new THREE.Vector3(P.casa.x + 0.55, 0, P.casa.z + 0.35);
  P.cammino = { punti: [...viaggio(o.position.clone(), P.stanza, alzati), P.casa.clone()], i: 0, sosta: 0 };
  P.seduto = false; P.fase = 0;
}
function aggiornaRiunioni(dati) {
  const dove = new Map();
  for (const Q of dati.riunioni || []) Q.partecipanti.forEach((a, i) => dove.set(a, { Q, i }));
  for (const [id, P] of Object.entries(S.agenti)) {
    const r = dove.get(id);
    if (r && P.riunione !== r.Q.id) vaiInRiunione(P, r.Q, r.i);
    else if (!r && P.riunione) tornaDaRiunione(P);
  }
  disegnaSchermoMeeting((dati.riunioni || []).find((Q) => Q.stanza === 'meeting') || null);
}

function passeggiate(s) {
  if (s < (S.prossima || 3)) return;
  S.prossima = s + 6 + Math.random() * 8;
  const inGiro = Object.values(S.agenti).filter((P) => P.cammino).length;
  if (inGiro >= 3) return;
  const liberi = Object.values(S.agenti).filter((P) => !P.cammino && !P.stato && !P.riunione && S.scelto !== P.A.id);
  if (!liberi.length) return;
  const P = liberi[Math.floor(Math.random() * liberi.length)];
  P.cammino = percorso(P); P.seduto = false; P.fase = 0;
}
function muovi(P, s, dt) {
  const C = P.cammino, o = P.U.o, obb = C.punti[C.i];
  if (obb === 'sosta') {
    if (P.G) glbMuovi(P.G, 'Idle', dt); else posaInPiedi(P.U, s);
    if (!C.sosta) C.sosta = s + 5 + Math.random() * 6;
    if (s > C.sosta) { C.i++; C.sosta = 0; }
    return;
  }
  const dir = obb.clone().sub(o.position); dir.y = 0;
  const dist = dir.length(), v = 1.25 * dt;
  if (dist < 0.05) {
    C.i++;
    if (C.i >= C.punti.length) {
      P.cammino = null; P.seduto = true;
      if (C.fine?.tipo === 'riunione') { o.position.copy(C.fine.seat.pos); o.rotation.y = C.fine.seat.rot; }
      else { o.position.copy(P.casa); o.rotation.y = 0; }
      if (P.G) glbSiedi(P.G);
    }
    return;
  }
  o.position.addScaledVector(dir.normalize(), Math.min(v, dist));
  const ang = Math.atan2(-dir.x, -dir.z);
  let d = ang - o.rotation.y; d = Math.atan2(Math.sin(d), Math.cos(d)); o.rotation.y += d * Math.min(1, dt * 10);
  P.fase += dt * 9; if (P.G) glbMuovi(P.G, 'Walk', dt); else posaCammino(P.U, P.fase);
}

// solo per le prove automatiche: inquadra un punto e fa camminare qualcuno
export function prova(cosa, a) {
  if (!S) return;
  if (cosa === 'vista') { S.volo = null; S.controls.target.set(a[3], a[4], a[5]); S.camera.position.set(a[0], a[1], a[2]); S.camera.lookAt(S.controls.target); }
  if (cosa === 'cammina') { const P = S.agenti[a]; P.cammino = percorso(P); P.seduto = false; P.fase = 0; }
  if (cosa === 'persone') return Object.values(S.agenti).filter((P) => P.G).length;
  if (cosa === 'salta') { const s = performance.now() / 1000; for (const P of Object.values(S.agenti)) for (let k = 0; k < 4000 && P.cammino; k++) { if (P.cammino.punti[P.cammino.i] === 'sosta') P.cammino.i++; else muovi(P, s, 0.05); } }
  if (cosa === 'sposta') { for (const P of Object.values(S.agenti)) if (P.G) { P.G.dz = a; P.G.w.position.z = a; } }
  if (cosa === 'dove') { const P = S.agenti[a]; return [P.U.o.position.x, P.U.o.position.z]; }
}

function anima(t) {
  if (!S) return;
  if (!document.body.contains(S.renderer.domElement)) { ferma(); return; }
  S.raf = requestAnimationFrame(anima);
  if (S.volo) {
    const v = S.volo, k = Math.min(1, (t - v.t0) / v.ms), e = k < 0.5 ? 4 * k * k * k : 1 - Math.pow(-2 * k + 2, 3) / 2;
    S.controls.target.lerpVectors(v.da, v.a, e);
    const dir = v.dir0.clone().lerp(v.dir1, e).normalize();
    S.camera.position.copy(S.controls.target).addScaledVector(dir, v.d0 + (v.d1 - v.d0) * e);
    if (k >= 1) S.volo = null;
  }
  S.controls.update();
  const s = t / 1000, dt = Math.min(0.2, s - (S.ultimo || s)); S.ultimo = s;
  for (const id in S.agenti) {
    const P = S.agenti[id];
    if (P.cammino) muovi(P, s, dt);
    else poseSeduto(P, s, P.stato === 'lavora' && !P.riunione, dt);
    if (P.stato === 'lavora' && s - (P.ts || 0) > 0.12) { P.ts = s; disegnaSchermo(P, s); P.sm.emissiveIntensity = 0.8 + Math.sin(s * 3 + P.x) * 0.1; }
  }
  passeggiate(s);
  if (S.cervello) { S.cervello.punti.rotation.y = s * 0.35; S.cervello.punti.rotation.x = Math.sin(s * 0.3) * 0.3; S.cervello.nucleo.material.emissiveIntensity = 0.7 + Math.sin(s * 2) * 0.25; }
  S.renderer.render(S.scena, S.camera);
  posiziona();
}

export function ferma() {
  if (!S) return;
  cancelAnimationFrame(S.raf);
  S.renderer.dispose(); S.controls.dispose();
  removeEventListener('resize', S.ridim);
  S = null;
}

function ridimensiona() {
  const r = S.cont.getBoundingClientRect();
  S.camera.aspect = r.width / r.height; S.camera.updateProjectionMatrix();
  S.renderer.setSize(r.width, r.height, false);
}

export function avvia(cont, dati, cb) {
  if (S) ferma();
  const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  renderer.shadowMap.enabled = true; renderer.shadowMap.type = THREE.PCFSoftShadowMap;
  renderer.toneMapping = THREE.ACESFilmicToneMapping; renderer.toneMappingExposure = 1.0;
  renderer.outputColorSpace = THREE.SRGBColorSpace;
  renderer.domElement.className = 'sala-canvas';
  cont.prepend(renderer.domElement);
  const scena = new THREE.Scene();
  scena.background = new THREE.Color('#f1f0ed');
  const pmrem = new THREE.PMREMGenerator(renderer);
  scena.environment = pmrem.fromScene(new RoomEnvironment(renderer), 0.04).texture;
  scena.environmentIntensity = 0.7;
  const camera = new THREE.PerspectiveCamera(30, 1, 0.1, 300);
  scena.add(new THREE.HemisphereLight(0xffffff, 0xe0dbd2, 0.45));
  const sole = new THREE.DirectionalLight(0xfffaf2, 2.2);
  sole.position.set(CX - 5, 20, CZ + 8); sole.target.position.set(CX, 0, CZ);
  sole.castShadow = true; sole.shadow.mapSize.set(4096, 4096); sole.shadow.bias = -0.0003; sole.shadow.normalBias = 0.02; sole.shadow.radius = 5;
  Object.assign(sole.shadow.camera, { left: -15, right: 15, top: 12, bottom: -12, near: 1, far: 70 });
  scena.add(sole, sole.target);
  const controls = new OrbitControls(camera, renderer.domElement);
  controls.enableDamping = true; controls.dampingFactor = 0.1; controls.screenSpacePanning = false;
  controls.minDistance = 3; controls.maxDistance = 60;
  controls.minPolarAngle = 0.35; controls.maxPolarAngle = 1.0; controls.minAzimuthAngle = -0.45; controls.maxAzimuthAngle = 0.45;
  controls.mouseButtons = { LEFT: THREE.MOUSE.PAN, MIDDLE: THREE.MOUSE.DOLLY, RIGHT: THREE.MOUSE.ROTATE };
  controls.touches = { ONE: THREE.TOUCH.PAN, TWO: THREE.TOUCH.DOLLY_ROTATE };
  const etichette = document.createElement('div'); etichette.className = 'etichette'; cont.appendChild(etichette);
  S = { renderer, scena, camera, controls, cont, cb: cb || {}, etichette, mat: materiali() };
  S.ridim = () => { if (S) ridimensiona(); }; addEventListener('resize', S.ridim);
  ridimensiona();
  costruisci(dati);
  caricaPersone().catch((e) => console.warn("persone 3D non caricate", e));
  grandangolo(true);
  const ray = new THREE.Raycaster(), pt = new THREE.Vector2(); let giu = null;
  renderer.domElement.addEventListener('pointerdown', (ev) => { giu = [ev.clientX, ev.clientY]; S.volo = null; });
  renderer.domElement.addEventListener('pointerup', (ev) => {
    if (!giu || Math.hypot(ev.clientX - giu[0], ev.clientY - giu[1]) > 6) return;
    const r = renderer.domElement.getBoundingClientRect();
    pt.set(((ev.clientX - r.left) / r.width) * 2 - 1, -((ev.clientY - r.top) / r.height) * 2 + 1);
    ray.setFromCamera(pt, camera);
    const hit = ray.intersectObjects(scena.children, true).find((h) => h.object.userData.ag || h.object.userData.uff || h.object.userData.cervello);
    if (!hit) return;
    if (hit.object.userData.cervello) S.cb.cervello && S.cb.cervello();
    else if (hit.object.userData.ag) vaiAgente(hit.object.userData.ag);
    else entraUfficio(hit.object.userData.uff);
  });
  S.raf = requestAnimationFrame(anima);
  return true;
}
